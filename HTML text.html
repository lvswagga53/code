<script>


function calculate() {


    const startBet = +document.getElementById("startBet").value;


    const rounds = +document.getElementById("rounds").value;


    const bankroll = +document.getElementById("bankroll").value;


    const profitTarget = +document.getElementById("profitTarget").value;


    const payout = +document.getElementById("payout").value;


    let totalLoss = 0;


    let profit = 0;


    let warning = "";


    let data = [];


    for (let i = 0; i < rounds; i++) {


        let bet = startBet * Math.pow(2, i);


        totalLoss += bet;


        let win = bet * (payout / 100);


        let netIfWin = win - totalLoss;


        profit = i * startBet; // simplified martingale cycle profit model


        data.push({


            round: i + 1,


            bet,


            loss: totalLoss,


            win,


            netIfWin


        });


        if (totalLoss > bankroll && !warning) {


            warning = "⚠️ Bankroll exceeded!";


        }


        if (profit >= profitTarget && !warning) {


            warning = "🎯 Profit target reached!";


        }


    }


    document.getElementById("warning").innerText = warning;


    drawChart(data, bankroll, profitTarget);


    renderTable(data);


}


function drawChart(data, bankroll, target) {


    const canvas = document.getElementById("chart");


    const ctx = canvas.getContext("2d");


    canvas.width = canvas.offsetWidth;


    canvas.height = 200;


    ctx.clearRect(0, 0, canvas.width, canvas.height);


    let max = Math.max(...data.map(d => d.loss), bankroll, target);


    // loss curve


    ctx.beginPath();


    data.forEach((d, i) => {


        let x = (i / (data.length - 1)) * canvas.width;


        let y = canvas.height - (d.loss / max) * canvas.height;


        if (i === 0) ctx.moveTo(x, y);


        else ctx.lineTo(x, y);


    });


    ctx.strokeStyle = "#22c55e";


    ctx.stroke();


    // bankroll line


    let y1 = canvas.height - (bankroll / max) * canvas.height;


    ctx.beginPath();


    ctx.moveTo(0, y1);


    ctx.lineTo(canvas.width, y1);


    ctx.strokeStyle = "#f87171";


    ctx.stroke();


    // target line


    let y2 = canvas.height - (target / max) * canvas.height;


    ctx.beginPath();


    ctx.moveTo(0, y2);


    ctx.lineTo(canvas.width, y2);


    ctx.strokeStyle = "#60a5fa";


    ctx.stroke();


}


function renderTable(data) {


    let html = `


    <table>


    <tr>


        <th>R</th>


        <th>Bet</th>


        <th>Loss</th>


        <th>Win</th>


        <th>Net</th>


    </tr>`;


    data.forEach(d => {


        html += `


        <tr>


            <td>${d.round}</td>


            <td>$${d.bet.toFixed(2)}</td>


            <td>$${d.loss.toFixed(2)}</td>


            <td>$${d.win.toFixed(2)}</td>


            <td style="color:${d.netIfWin >= 0 ? '#22c55e' : '#f87171'}">


                $${d.netIfWin.toFixed(2)}


            </td>


        </tr>`;


    });


    html += "</table>";


    document.getElementById("tableBox").innerHTML = html;


}
